   50: def cluster_quota_select(
   51:     candidates: pd.DataFrame,
   52:     *,
   53:     k: int,
   54:     cluster_column: str,
   55:     score_column: str = "score",
   56:     quota: str = "sqrt_size",
   57:     caps: Mapping[str, int] | None = None,
   58:     seed: int = 0,
   59:     within_cluster: str = "top",
   60: ) -> pd.DataFrame:
   61:     if cluster_column not in candidates:
   62:         raise SchemaError(f"cluster quota selection requires {cluster_column}")
   63:     if k > len(candidates):
   64:         raise BudgetError("requested unique count exceeds candidate pool")
   65:     groups = list(candidates.groupby(cluster_column, dropna=False, sort=True))
   66:     sizes = np.asarray([len(group) for _, group in groups], dtype=float)
   67:     if quota == "equal":
   68:         weights = np.ones_like(sizes)
   69:     elif quota == "proportional":
   70:         weights = sizes
   71:     elif quota == "sqrt_size":
   72:         weights = np.sqrt(sizes)
   73:     else:
   74:         raise ValueError("quota must be equal, proportional, or sqrt_size")
   75:     raw = weights / weights.sum() * k
   76:     target = np.floor(raw).astype(int)
   77:     target = np.minimum(target, sizes.astype(int))
   78:     remaining = k - int(target.sum())
   79:     while remaining > 0:
   80:         capacity = sizes.astype(int) - target
   81:         available = np.flatnonzero(capacity > 0)
   82:         if len(available) == 0:
   83:             raise BudgetError("cluster quotas cannot fill requested unique count")
   84:         priority = (raw - target)[available]
   85:         idx = available[int(np.argmax(priority))]
   86:         target[idx] += 1
   87:         remaining -= 1
   88: 
   89:     rng = np.random.default_rng(seed)
   90:     parts: list[pd.DataFrame] = []
   91:     for (cluster, group), n in zip(groups, target, strict=True):
   92:         if n == 0:
   93:             continue
   94:         if within_cluster == "top":
   95:             local = group.sort_values([score_column, "sample_id"], ascending=[False, True]).head(int(n))
   96:         elif within_cluster == "probability":
   97:             score = group[score_column].to_numpy(float)
   98:             weight = np.exp((score - np.max(score)) / max(np.std(score), 1e-6))
   99:             idx = rng.choice(len(group), size=int(n), replace=False, p=weight / weight.sum())
  100:             local = group.iloc[idx]
  101:         elif within_cluster == "random":
  102:             local = group.iloc[rng.choice(len(group), size=int(n), replace=False)]
  103:         else:
  104:             raise ValueError("within_cluster must be top, probability, or random")
  105:         parts.append(local)
  106:     provisional = pd.concat(parts, ignore_index=True)
  107:     # Reapply caps globally in score order; fill deficits from the full pool.
  108:     ordered = pd.concat(
  109:         [
  110:             provisional.sort_values([score_column, "sample_id"], ascending=[False, True]),
  111:             candidates.sort_values([score_column, "sample_id"], ascending=[False, True]),
  112:         ],
  113:         ignore_index=True,
  114:     ).drop_duplicates("sample_id", keep="first")

  173: def build_snapshot_schedule(
  174:     snapshots: Sequence[tuple[int, pd.DataFrame]],
  175:     *,
  176:     epochs: int,
  177:     seed: int = 0,
  178: ) -> ReplaySchedule:
  179:     """Build an offline dynamic schedule from selections activated at 1-indexed epochs.
  180: 
  181:     Every snapshot must have the same replay-slot budget. This preserves epoch length,
  182:     optimizer steps, and LR schedule across T/R1/R2 arms.
  183:     """
  184:     if not snapshots:
  185:         raise BudgetError("at least one schedule snapshot is required")
  186:     ordered = sorted((int(start), validate_selection(table)) for start, table in snapshots)
  187:     if ordered[0][0] != 1:
  188:         raise BudgetError("the first snapshot must start at epoch 1; use a matched-random warm-up snapshot if needed")
  189:     budgets = {int(table["repeat_count"].sum()) for _, table in ordered}
  190:     if len(budgets) != 1:
  191:         raise BudgetError(f"all snapshots must have the same per-epoch slot budget, got {sorted(budgets)}")
  192:     slots = budgets.pop()
  193:     all_ids = np.asarray(sorted({str(sid) for _, table in ordered for sid in table["sample_id"]}))
  194:     lookup = {sid: i for i, sid in enumerate(all_ids)}
  195:     matrix = np.empty((epochs, slots), dtype=np.int32)
  196:     snapshot_starts = [start for start, _ in ordered]
  197:     for epoch in range(1, epochs + 1):
  198:         active_idx = max(i for i, start in enumerate(snapshot_starts) if start <= epoch)
  199:         expanded = _expand_selection(ordered[active_idx][1])
  200:         row = np.asarray([lookup[sid] for sid in expanded], dtype=np.int32)
  201:         np.random.default_rng(child_seed(seed, "snapshot", epoch)).shuffle(row)
  202:         matrix[epoch - 1] = row
  203:     return ReplaySchedule(
  204:         all_ids,
  205:         matrix,
  206:         np.arange(1, epochs + 1),
  207:         {
  208:             "budget_unit": "append_slots_per_epoch",
  209:             "selection_slot_budget": slots,
  210:             "expected_total_scheduled_exposures": slots * epochs,
  211:             "schedule_seed": int(seed),
  212:             "schedule_type": "offline_snapshots",
  213:             "snapshot_start_epochs": snapshot_starts,
  214:         },
  215:     )

===== docs/METHODS_AND_LITERATURE.md =====
   61: ## 3. Layer B — slow-learning feature groups
   62: 
   63: ### Source
   64: 
   65: Nguyen et al., *USEFUL: A Framework for Unified Supervised and Unsupervised Feature Learning*, NeurIPS 2024, as analyzed in the included second-round review.
   66: 
   67: ### Source method
   68: 
   69: USEFUL identifies an early slow-learnable group and physically over-samples the original samples. Its reported image-classification experiments distinguish slow-learning groups from merely highest-loss or currently misclassified samples, and include random-over-sampling comparisons and multiplier/timing ablations.
   70: 
   71: ### Binary-task adaptation
   72: 
   73: A two-class softmax output has only one independent probability. Clustering one final probability would reduce to another confidence threshold. Therefore `add_slow_learning_clusters` uses:
   74: 
   75: - loss trajectory vectors by default;
   76: - optionally representation or gradient-embedding columns;
   77: - class-conditional clustering;
   78: - the cluster with higher probe/mean loss as the slow group.
   79: 
   80: Code:
   81: 
   82: ```text
   83: stage1_replay/dynamics/trajectory.py
   84: configs/methods/slow_learning.yaml
   85: ```
   86: 
   87: ### Non-claim
   88: 
   89: This is a Budgeted-USEFUL adaptation, not the authors' original implementation. The slow group can still contain noise, so it is separately intersected with reducibility and reliability.
   90: 
   91: ## 4. Layer C — cross-fitted residual learnability
   92: 
   93: ### Sources
   94: 
   95: - Mindermann et al., *Prioritized Training on Points that are Learnable, Worth Learning, and Not Yet Learnt*, ICML 2022 (RHO-LOSS).
   96: - ReLo: reducible-loss replay priority.
   97: - CSRL: reducible-loss replay/coreset selection.
   98: - the included second- and third-round reviews for exact distinctions and limitations.
   99: 
  100: ### Formula
  101: 
  102: When a current-run checkpoint exists:
  103: 
  104: \[
  105: R_i(t)=\left[z_g(L_i^{cur}(t))-z_g(L_i^{OOF-ref})\right]_+.
  106: \]
  107: 
  108: A percentile alternative is also recorded:
  109: 
  110: \[
  111: R_i^{pct}(t)=\left[Q_g(L_i^{cur}(t))-Q_g(L_i^{OOF-ref})\right]_+.
  112: \]
  113: 
  114: Here `g` can be class, source, fold or business-channel strata.
  115: 
  116: When no current-run checkpoint table is provided, the package can use an early/probe OOF loss against late OOF reference loss. It marks:
  117: 
  118: ```text
  119: reducible_is_proxy = 1
  120: ```
  121: 
  122: so historical proxy runs cannot be silently mixed with true current-state runs.
  123: 
  124: ### Terminology guard
  125: 
  126: The code uses:
  127: 
  128: ```text
  129: oof_reference_loss
  130: cross-fitted reference predictability
  131: ```
  132: 
  133: It does not call this term true irreducible loss, because OOF models can still have capacity, calibration and fold-distribution errors.

   14: def _long_table() -> pd.DataFrame:
   15:     rows = []
   16:     for trajectory in ["fold-0", "fold-1"]:
   17:         for sample_id, label in [("a", 0), ("b", 1), ("c", 0)]:
   18:             for epoch in [1, 2, 3, 4]:
   19:                 base = 0.1 * epoch + (0.05 if trajectory == "fold-1" else 0.0)
   20:                 p_defect = base if label == 1 else 1.0 - base
   21:                 rows.append(
   22:                     {
   23:                         "sample_id": sample_id,
   24:                         "label": label,
   25:                         "epoch": epoch,
   26:                         "trajectory_id": trajectory,
   27:                         "p_defect": p_defect,
   28:                     }
   29:                 )
   30:     return pd.DataFrame(rows)
